/* Code.gs - Google Apps Script Server Backend */
const SPREADSHEET_ID = "REPLACE_WITH_YOUR_GOOGLE_SHEET_ID";

function doGet(e) {
  return HtmlService.createTemplateFromFile('Index')
    .evaluate()
    .setTitle('DriveFusion Multi-Cloud Storage')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1.0');
}

function setupDatabase() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  
  // Sheet Users
  let userSheet = ss.getSheetByName('Users') || ss.insertSheet('Users');
  if(userSheet.getLastRow() === 0) {
    userSheet.appendRow(['UserID', 'Username', 'Email', 'Role', 'CreatedAt']);
  }
  
  // Sheet LinkedDrives
  let driveSheet = ss.getSheetByName('LinkedDrives') || ss.insertSheet('LinkedDrives');
  if(driveSheet.getLastRow() === 0) {
    driveSheet.appendRow(['DriveID', 'AccountEmail', 'AccountName', 'TotalQuotaGB', 'UsedQuotaGB', 'Status']);
  }
  
  // Sheet FileIndex
  let fileSheet = ss.getSheetByName('FileIndex') || ss.insertSheet('FileIndex');
  if(fileSheet.getLastRow() === 0) {
    fileSheet.appendRow(['FileID', 'FileName', 'FileType', 'FileSizeMB', 'DriveID', 'ParentFolderID', 'CreatedAt', 'IsTrash']);
  }
  
  return "Database Setup Complete!";
}

// Smart Storage Allocation Engine
function getBestDriveForUpload(fileSizeBytes) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName('LinkedDrives');
  const data = sheet.getDataRange().getValues();
  
  let bestDrive = null;
  let maxFreeSpace = -1;
  
  for (let i = 1; i < data.length; i++) {
    const driveId = data[i][0];
    const totalGB = parseFloat(data[i][3]);
    const usedGB = parseFloat(data[i][4]);
    const freeGB = totalGB - usedGB;
    
    if (freeGB > maxFreeSpace) {
      maxFreeSpace = freeGB;
      bestDrive = driveId;
    }
  }
  return bestDrive;
}
