// =================================================================
// DRIVEFUSION - GOOGLE APPS SCRIPT BACKEND ENGINE (Code.gs)

const SPREADSHEET_ID = "GANTI_DENGAN_ID_GOOGLE_SHEETS_ANDA";

function doGet(e) {
  return HtmlService.createTemplateFromFile('Index')
    .evaluate()
    .setTitle('DriveFusion - Unified Multi-Cloud Storage')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1.0')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

// 1. Ambil Semua File & Folder Index dari Database Sheets
function getUnifiedFileSystem() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName('FileIndex');
  const data = sheet.getDataRange().getValues();
  
  if (data.length <= 1) return [];
  const headers = data[0];
  
  return data.slice(1).map(row => ({
    id: row[0],
    name: row[1],
    type: row[2], // 'folder' | 'file'
    category: row[3],
    size: row[4],
    driveId: row[5],
    driveEmail: row[6],
    parentId: row[7],
    updatedAt: row[8]
  }));
}

// 2. Smart Auto-Allocator: Cari Akun Drive Dengan Sisa Ruang Terbesar
function getSmartDriveTarget() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName('LinkedDrives');
  const data = sheet.getDataRange().getValues();
  
  let bestDrive = null;
  let maxFree = -1;
  
  for (let i = 1; i < data.length; i++) {
    const freeSpace = data[i][3] - data[i][2]; // Total - Used
    if (freeSpace > maxFree) {
      maxFree = freeSpace;
      bestDrive = {
        id: data[i][0],
        label: data[i][1],
        usedGB: data[i][2],
        totalGB: data[i][3],
        freeGB: freeSpace.toFixed(1),
        email: data[i][4]
      };
    }
  }
  return bestDrive;
}

// 3. Simpan File Baru (Simulasi/DriveApp Integration)
function saveFileToOptimalDrive(fileMetaData) {
  const targetDrive = getSmartDriveTarget();
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const indexSheet = ss.getSheetByName('FileIndex');
  
  const newFileId = "FILE_" + Utilities.getUuid();
  indexSheet.appendRow([
    newFileId,
    fileMetaData.name,
    'file',
    fileMetaData.category,
    fileMetaData.size,
    targetDrive.id,
    targetDrive.email,
    fileMetaData.parentId || 'root',
    new Date().toISOString()
  ]);
  
  return { success: true, allocatedDrive: targetDrive, fileId: newFileId };
}

function setupDatabase() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // 1. Tabel Master User
  let userSheet = ss.getSheetByName('Users') || ss.insertSheet('Users');
  userSheet.clear();
  userSheet.appendRow(['UserID', 'Email', 'FullName', 'Role', 'PasswordHash', 'CreatedAt']);
  userSheet.appendRow(['USR_01', 'master.admin@cloud.corp', 'Enterprise Administrator', 'Admin', 'HASH_SECRET_123', new Date()]);
  
  // 2. Tabel Linked Drive Accounts
  let driveSheet = ss.getSheetByName('LinkedDrives') || ss.insertSheet('LinkedDrives');
  driveSheet.clear();
  driveSheet.appendRow(['DriveID', 'DriveLabel', 'UsedCapacityGB', 'TotalCapacityGB', 'DriveEmail', 'Status']);
  driveSheet.appendRow(['DRV_1', 'Drive #1 (Primary Work)', 10.2, 15.0, 'john.primary@gmail.com', 'Active']);
  driveSheet.appendRow(['DRV_2', 'Drive #2 (Project Archive)', 2.2, 15.0, 'john.work@gmail.com', 'Active']);
  driveSheet.appendRow(['DRV_3', 'Drive #3 (Media Vault)', 4.8, 15.0, 'media.john@gmail.com', 'Active']);
  driveSheet.appendRow(['DRV_4', 'Drive #4 (Backup Data)', 1.2, 15.0, 'backup.john@gmail.com', 'Active']);

  // 3. Tabel File Index Multi-Drive Map
  let indexSheet = ss.getSheetByName('FileIndex') || ss.insertSheet('FileIndex');
  indexSheet.clear();
  indexSheet.appendRow(['FileID', 'FileName', 'FileType', 'Category', 'SizeFormatted', 'DriveID', 'DriveEmail', 'ParentID', 'UpdatedAt']);
  
  Logger.log("✅ Database DriveFusion Sukses Dibuat di Google Sheets!");
}
